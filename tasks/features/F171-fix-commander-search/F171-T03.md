### Test 1: Commander search with bare cards (reproduces the bug)

Add to `tests/unit/decks/test_commander_search.py`:

```python
class TestCommanderSearchBareCards:
    """Reproduce F171: cards created by price collector have NULL type_line."""

    def test_bare_card_not_found_by_commander_search(self, repo):
        """A card with type_line=NULL is correctly excluded from commander search."""
        _add(repo, CardRow(
            game="magic",
            name_en="Atraxa, Praetors' Voice",
            set_code="cmm",
            collector_number="1",
            # No type_line, rarity, color_identity — simulates upsert_card
        ))
        result = get_commander_candidates(repo, search="atraxa")
        assert result == []  # Expected: bug behavior before fix

    def test_enriched_card_found_by_commander_search(self, repo):
        """After enrichment, the same card appears in commander search."""
        with Session(repo.engine) as session:
            card = CardRow(
                game="magic",
                name_en="Atraxa, Praetors' Voice",
                set_code="cmm",
                collector_number="1",
            )
            session.add(card)
            session.commit()

            # Simulate enrichment (what T01 fix does)
            card_id = card.id
            session.execute(
                update(CardRow)
                .where(CardRow.id == card_id)
                .values(
                    type_line="Legendary Creature — Phyrexian Angel",
                    color_identity="WUBG",
                    rarity="mythic",
                )
            )
            session.commit()

        result = get_commander_candidates(repo, search="atraxa")
        assert len(result) == 1
        assert result[0]["name_en"] == "Atraxa, Praetors' Voice"
```
