### 2. Test config priority

File: `tests/unit/test_config.py`

```python
def test_database_url_takes_priority(monkeypatch):
    monkeypatch.setenv("DATABASE_URL", "postgresql://neon")
    monkeypatch.setenv("TCG_DATABASE_URL", "sqlite:///old.db")
    assert get_db_url() == "postgresql://neon"

def test_tcg_database_url_fallback(monkeypatch):
    monkeypatch.delenv("DATABASE_URL", raising=False)
    monkeypatch.setenv("TCG_DATABASE_URL", "sqlite:///custom.db")
    assert get_db_url() == "sqlite:///custom.db"

def test_default_sqlite(monkeypatch):
    monkeypatch.delenv("DATABASE_URL", raising=False)
    monkeypatch.delenv("TCG_DATABASE_URL", raising=False)
    assert get_db_url() == "sqlite:///tcg_market.db"
```
