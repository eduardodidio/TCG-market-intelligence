### `src/services/news_fetcher.py`

```python
import feedparser  # new dependency — needs user confirmation
import structlog
from datetime import datetime
from src.database.repository import Repository

logger = structlog.get_logger(__name__)

SOURCES = [
    {
        "name": "mtg_official",
        "url": "https://magic.wizards.com/en/news/rss.xml",
        "category_default": "release",
    },
    {
        "name": "scryfall_blog",
        "url": "https://scryfall.com/blog/rss",
        "category_default": "other",
    },
]

def fetch_news(repo: Repository, max_per_source: int = 20) -> dict:
    """Fetch news from all configured RSS sources. Returns stats."""
    stats = {"fetched": 0, "new": 0, "skipped": 0, "errors": []}

    for source in SOURCES:
        try:
            feed = feedparser.parse(source["url"])
            for entry in feed.entries[:max_per_source]:
                stats["fetched"] += 1
                # Extract fields
                item = {
                    "title": entry.get("title", "Untitled"),
                    "summary": _clean_summary(entry.get("summary", "")),
                    "source_url": entry.get("link", ""),
                    "source_name": source["name"],
                    "category": _categorize(entry, source["category_default"]),
                    "image_url": _extract_image(entry),
                    "published_at": _parse_date(entry),
                }
                if not item["source_url"]:
                    stats["skipped"] += 1
                    continue
                inserted = repo.upsert_news_item(item)
                if inserted:
                    stats["new"] += 1
                else:
                    stats["skipped"] += 1
        except Exception as e:
            logger.error("news_fetch_error", source=source["name"], error=str(e))
            stats["errors"].append(f"{source['name']}: {e}")

    return stats
```

### CLI command in `src/cli/main.py`

```python
@cli.command()
@click.option("--max-per-source", default=20, help="Max items per source")
def fetch_news(max_per_source):
    """Fetch MTG news from RSS sources."""
    repo = get_repository()
    stats = news_fetcher.fetch_news(repo, max_per_source)
    click.echo(f"Fetched: {stats['fetched']}, New: {stats['new']}, Skipped: {stats['skipped']}")
    if stats['errors']:
        for e in stats['errors']:
            click.echo(f"  [ERROR] {e}", err=True)
```

### `bats/fetch-news.bat`

```bat
@echo off
REM ============================================================
REM  fetch-news.bat — Busca noticias de MTG de fontes RSS
REM  Rodar manualmente quando quiser atualizar o feed
REM ============================================================

cd /d "%~dp0\.."

echo ============================================================
echo  TEDHC Fetch News - %date% %time%
echo ============================================================

python -m src.cli.main fetch-news --max-per-source 20

echo.
echo  [DONE] %date% %time%
echo ============================================================
```
