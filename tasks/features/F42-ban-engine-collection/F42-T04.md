### API client (`frontend/src/api/banEngine.ts`)

```typescript
import { apiGet } from "./client";
import type { ApiResponse, BannedCollectionCard, CardLegalityWithChange } from "../types/api";

export function fetchCollectionBanned(
  params?: { format?: string; days?: number }
): Promise<ApiResponse<BannedCollectionCard[]>> {
  const query = new URLSearchParams();
  if (params?.format) query.set("format", params.format);
  if (params?.days) query.set("days", String(params.days));
  const qs = query.toString();
  return apiGet(`/collection/banned${qs ? `?${qs}` : ""}`);
}

export function fetchCardLegalities(
  entryId: number,
  params?: { days?: number }
): Promise<ApiResponse<CardLegalityWithChange[]>> {
  const query = new URLSearchParams();
  if (params?.days) query.set("days", String(params.days));
  const qs = query.toString();
  return apiGet(`/collection/${entryId}/legality${qs ? `?${qs}` : ""}`);
}
```

Uses `apiGet` from existing `client.ts` (includes auth token automatically).
