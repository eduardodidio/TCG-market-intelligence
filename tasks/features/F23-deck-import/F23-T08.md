# F23-T08: Frontend Types + API Client

- **Wave:** 4
- **Status:** planned
- **Depends on:** F23-T07
- **Type:** frontend

## Description

Add TypeScript types and API client functions for the deck feature.

### Types (in `frontend/src/types/api.ts`)

```typescript
export interface DeckSummary {
  id: number;
  name: string;
  description: string | null;
  total_cards: number;
  unique_cards: number;
  owned_cards: number;
  ownership_pct: number;
  created_at: string;
  updated_at: string;
}

export interface DeckCard {
  id: number;
  name_en: string;
  set_code: string | null;
  collector_number: string | null;
  quantity: number;
  card_id: number | null;
  in_collection: boolean;
  owned_quantity: number;
  collection_entry_id: number | null;
  image_url: string | null;
  latest_price: number | null;
}

export interface DeckDetail {
  id: number;
  name: string;
  description: string | null;
  cards: DeckCard[];
  total_cards: number;
  unique_cards: number;
  owned_cards: number;
  ownership_pct: number;
  created_at: string;
  updated_at: string;
}

export interface DeckImportResult {
  deck_id: number;
  name: string;
  cards_imported: number;
  cards_linked: number;
}
```

### API Client (`frontend/src/api/decks.ts`)

```typescript
export function fetchDecks(): Promise<ApiResponse<DeckSummary[]>>
export function fetchDeck(id: number): Promise<ApiResponse<DeckDetail>>
export function importDeck(name: string, format: "text" | "csv", content: string): Promise<ApiResponse<DeckImportResult>>
export function deleteDeck(id: number): Promise<void>
```

Follow the same pattern as `frontend/src/api/collection.ts`.
