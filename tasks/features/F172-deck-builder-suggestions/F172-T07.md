## Implementation details

```ts
const BASE = "/api/v1/deck-suggestions";
export function createDeckSuggestion(body: DeckSuggestionCreate) { return apiPost<DeckSuggestion>(BASE, body); }
export function listDeckSuggestions(status?: SuggestionStatus) { return apiGet<DeckSuggestion[]>(BASE, status ? { status } : undefined); }
export function getDeckSuggestion(id: number) { return apiGet<DeckSuggestion>(`${BASE}/${id}`); }
export function saveDeckSuggestion(id: number, deckName?: string) { return apiPost<{ deck_id: number }>(`${BASE}/${id}/save`, deckName ? { deck_name: deckName } : {}); }
export function deleteDeckSuggestion(id: number) { return apiDelete(`${BASE}/${id}`); }
```

Also export constants shared by the components: `SUGGESTION_FORMATS` (value + label + card count),
`SUGGESTION_ARCHETYPES` (aggro, control, midrange, combo, tempo, ramp with a pt-BR/English label default), and `MTG_COLOR_KEYS = ["W","U","B","R","G"]`.
